if (healthyOn && guardOn) {
    return {
      label: "Ready",
      className: "safe",
    };
  }

  if (!healthyOn && !guardOn) {
    return {
      label: "Guard open",
      className: "warning",
    };
  }

  if (!healthyOn && guardOn) {
    return {
      label: "Guard Open",
      className: "warning",
    };
  }

  if (healthyOn && !guardOn) {
    return {
      label: "Fault",
      className: "danger",
    };
  }

  return {
    label: "Unknown",
    className: "warning",
  };
}
